\begin{inlineexercise}{Checkpoint}{}{section-not-printout-3}%
%
\begin{itemize}[label=\ptxlistdisc]
\item{}Inside inline math (short, space for \(x\)): \(\sin(\fillinmath{x})\)%
\item{}Inside inline math (default, space for \(XXX\)): \(\sin(\fillinmath{XXX})\)%
\item{}Inside exponents and subscripts (each is space for the string \textquotedblleft{}12\textquotedblright{}).  In this case, be sure to wrap your exponents and subscripts in braces, as would be good \LaTeX{} practice anyway: \(x^{5+\fillinmath{12}}\,y_{\fillinmath{12}}\)%
\item{}Inside inline math (too long for this line probably, 40 characters long): \(\tan(\fillinmath{Mxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxxx})\)%
\item{}So use inside a displayed equation%
\begin{equation*}
16\log\space\fillinmath{1-x^2}
\end{equation*}
like this one.%
\item{}Inside the second line of a multi-line display:%
\begin{align*}
y &= x^7\,x^8\\
&= x^{\fillinmath{15}}
\end{align*}
%
\item{}This fillin has the historical \ptxmono{@characters} attribute for a fillin inside math: \(1+\fillinmath{XXXXX}+4=10\), which may be more convenient, but may not side properly in places like subscripts, superscripts, fractions, limits of integrals, and so on.%
\end{itemize}
%
\end{inlineexercise}%
